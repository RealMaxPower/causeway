  <L2 title="The shape behind the number">
    <L2Grid
      rail={
        <SideRail title="Key terms">
          <KeyTerm term="Lorenz curve">
            Cumulative income share against cumulative population share.
            Diagonal = perfect equality; bowed-out = inequality.
          </KeyTerm>
          <KeyTerm term="Gini coefficient">
            Area between the equality line and the Lorenz curve, divided
            by the triangle. 0 = perfect equality; 1 = one person has it
            all. Single number, lossy summary.
          </KeyTerm>
          <KeyTerm term="Decile share">
            Share of total income going to each 10% slice of the
            population. D1 = bottom 10%; D10 = top 10%.
          </KeyTerm>
          <KeyTerm term="Top 1% share">
            What the top 1% of earners take. Often the most politically
            salient number; not what Gini emphasises.
          </KeyTerm>
          <KeyTerm term="Median / mean ratio">
            Median income ÷ mean. 1 = symmetric distribution. Lower =
            right-skewed (a few very-rich pull the mean above the typical).
          </KeyTerm>
        </SideRail>
      }
    >
      <L2Step num="01" title="Inequality is a shape">
